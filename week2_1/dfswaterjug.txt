JUGS = [8, 5, 3]
GOAL = 4
start_state = (0, 0, 0)

def get_next_states(state):
    next_states = []
    a, b, c = state
    current = [a, b, c]
    for i in range(3):
        fill = current[:]
        fill[i] = JUGS[i]
        next_states.append(tuple(fill))
        empty = current[:]
        empty[i] = 0
        next_states.append(tuple(empty))
        for j in range(3):
            if i != j:
                pour_amount = min(current[i], JUGS[j] - current[j])
                if pour_amount > 0:
                    new_state = current[:]
                    new_state[i] -= pour_amount
                    new_state[j] += pour_amount
                    next_states.append(tuple(new_state))
    return next_states

def dfs(start):
    stack = [(start, [start])]
    visited = set()
    visited.add(start)
    while stack:
        state, path = stack.pop()
        if GOAL in state:
            return path
        for next_state in get_next_states(state):
            if next_state not in visited:
                visited.add(next_state)
                stack.append((next_state, path + [next_state]))
    return None

dfs_path = dfs(start_state)
print("DFS Solution (any path):")
for step, s in enumerate(dfs_path):
    print(f"Step {step}: {s}")
print(f"Total moves: {len(dfs_path)-1}")
